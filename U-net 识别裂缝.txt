import cv2

IMG_SIZE_DEFAULT = 512
SAVE_PANELS = 8


# ------------------------------------------------------------------ 1. 配置
def parse_args():
    ap = argparse.ArgumentParser()
    ap.add_argument("--data", default=r"E:\class2", help="PaveCrack1300 根目录（含 images/ 与 masks/）")
    ap.add_argument("--out", default="run_pavecrack", help="输出目录（别写进数据集文件夹）")
    ap.add_argument("--size", type=int, default=IMG_SIZE_DEFAULT, help="必须是 16 的倍数")
    ap.add_argument("--batch", type=int, default=4)
    ap.add_argument("--epochs", type=int, default=30)
    ap.add_argument("--lr", type=float, default=2e-4)
    ap.add_argument("--workers", type=int, default=0, help="Windows 保持 0 最省事")
    ap.add_argument("--seed", type=int, default=42)
    ap.add_argument("--no-aug", action="store_true", help="关掉增强，用于对照实验")
    a, _ = ap.parse_known_args() 
    assert a.size % 16 == 0, "尺寸必须是 16 的倍数（网络下采样 4 次）"
    return a


def set_seed(seed):
    random.seed(seed); np.random.seed(seed); torch.manual_seed(seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(seed)


# ------------------------------------------------------------------ 2. 配对与体检
def collect_pairs(root: Path) -> List[Dict]:
    img_dir, msk_dir = root / "images", root / "masks"
    if not img_dir.is_dir() or not msk_dir.is_dir():
        raise FileNotFoundError("找不到数据目录：\n%s\n%s" % (img_dir, msk_dir))

    imgs = sorted(p for p in img_dir.iterdir() if p.suffix.lower() in {".jpg", ".jpeg"})
    mask_by_stem = {p.stem: p for p in msk_dir.iterdir() if p.suffix.lower() == ".png"}

    meta_ratio = {}
    meta_csv = root / "sample_metadata.csv"
    if meta_csv.is_file():
        with meta_csv.open("r", encoding="utf-8-sig", newline="") as f:
            for row in csv.DictReader(f):
                name = (row.get("image_file") or "").strip()
                raw = (row.get("crack_pixel_ratio") or "").strip()
                if name and raw:
                    try:
                        meta_ratio[Path(name).stem] = float(raw)
                    except ValueError:
                        pass
        if meta_ratio:
            # 比例可能是 0~1 小数，也可能是 0~100 百分数，统一成百分数
            vmax = max(meta_ratio.values())
            if vmax <= 1.0:
                meta_ratio = {k: v * 100.0 for k, v in meta_ratio.items()}
                print("[体检] metadata 的 crack_pixel_ratio 是 0~1 小数，已换算成百分数")

    pairs, missing = [], []
    for p in imgs:
        mp = mask_by_stem.get(p.stem)
        if mp is None:
            missing.append(p.name)
            continue
        ratio = meta_ratio.get(p.stem)
        if ratio is None:
            m = cv2.imread(str(mp), cv2.IMREAD_GRAYSCALE)
            if m is None:
                raise ValueError("掩膜读取失败：" + str(mp))
            ratio = float(np.count_nonzero(m > 0) / m.size * 100.0)
        # 按裂缝占比分层：0–5 / 5–10 / 10–20 / >20
        pairs.append({"image": p, "mask": mp, "ratio": ratio,
                      "bin": int(np.digitize(ratio, bins=[5.0, 10.0, 20.0], right=True))})

    if missing:
        raise FileNotFoundError("有 %d 张原图没有同名 PNG 掩膜，例如：%s"
                                % (len(missing), ", ".join(missing[:8])))
    if not pairs:
        raise RuntimeError("没有配到任何 原图 / 掩膜 对")

    # 逐张体检：尺寸一致、掩膜是 0/255，避免"悄悄缩放"和"标签读错"
    for r in pairs:
        im = cv2.imread(str(r["image"]), cv2.IMREAD_COLOR)
        mk = cv2.imread(str(r["mask"]), cv2.IMREAD_GRAYSCALE)
        if im is None or mk is None:
            raise ValueError("读取失败：%s" % r["image"])
        if im.shape[:2] != mk.shape[:2]:
            raise ValueError("原图与掩膜尺寸不一致：%s" % r["image"].name)
        vals = set(np.unique(mk).tolist())
        if not vals.issubset({0, 255}):
            raise ValueError("掩膜不是 0/255 二值：%s，取值=%s" % (r["mask"].name, sorted(vals)[:10]))
    print("[体检] 共 %d 对，尺寸与掩膜取值全部通过" % len(pairs))
    if len(pairs) != 1300:
        print("[提示] README 记载 1300 对，当前 %d 对，请核对解压目录" % len(pairs))
    return pairs


# ------------------------------------------ 3. 分层划分（纯 Python，不依赖 sklearn）
def stratified_split(pairs: Sequence[Dict], seed=42):
    rng = random.Random(seed)
    by_bin: Dict[int, List[Dict]] = {}
    for r in pairs:
        by_bin.setdefault(r["bin"], []).append(r)
    tr, va, te = [], [], []
    total = sum(len(v) for v in by_bin.values())
    cum = 0                              # 累计取整：保证三段合计正好 70/10/20
    for b in sorted(by_bin):
        items = list(by_bin[b]); rng.shuffle(items)
        cum_prev, cum = cum, cum + len(items)
        n_te = int(round(cum * 0.20)) - int(round(cum_prev * 0.20))
        n_va = int(round(cum * 0.10)) - int(round(cum_prev * 0.10))
        n_tr = len(items) - n_te - n_va
        if len(items) >= 20:             # 样本多的箱内三段都非空
            n_te, n_va = max(n_te, 1), max(n_va, 1)
            n_tr = len(items) - n_te - n_va
        tr += items[:n_tr]; va += items[n_tr:n_tr + n_va]; te += items[n_tr + n_va:]
    assert total == len(tr) + len(va) + len(te)
    for x in (tr, va, te):
        x.sort(key=lambda r: r["image"].name)
    return tr, va, te


def save_split(tr, va, te, out_dir: Path):
    with (out_dir / "data_split.csv").open("w", encoding="utf-8-sig", newline="") as f:
        w = csv.writer(f)
        w.writerow(["split", "image_file", "mask_file", "crack_ratio_percent", "density_bin"])
        for tag, recs in (("train", tr), ("val", va), ("test", te)):
            for r in recs:
                w.writerow([tag, r["image"].name, r["mask"].name, "%.4f" % r["ratio"], r["bin"]])


# ------------------------------------------------------------------ 4. 数据集与同步增强
class CrackDataset(Dataset):
    def __init__(self, records, size=512, augment=False):
        self.records, self.size, self.augment = list(records), size, augment

    def __len__(self):
        return len(self.records)

    def __getitem__(self, i):
        r = self.records[i]
        im = cv2.imread(str(r["image"]), cv2.IMREAD_COLOR)      # 模型第一层 3 通道，必须读彩色
        mk = cv2.imread(str(r["mask"]), cv2.IMREAD_GRAYSCALE)
        if im is None or mk is None:
            raise RuntimeError("读取失败：%s / %s" % (r["image"], r["mask"]))
        im = cv2.cvtColor(im, cv2.COLOR_BGR2RGB)
        mk = (mk > 0).astype(np.uint8)                          # 255/0 → 1/0

        if (im.shape[0], im.shape[1]) != (self.size, self.size):
            im = cv2.resize(im, (self.size, self.size), interpolation=cv2.INTER_AREA)
            mk = cv2.resize(mk, (self.size, self.size), interpolation=cv2.INTER_NEAREST)

        if self.augment:                                        # 几何变换图与标同做，亮度只动图
            if random.random() < 0.5:
                im, mk = cv2.flip(im, 1), cv2.flip(mk, 1)
            if random.random() < 0.5:
                im, mk = cv2.flip(im, 0), cv2.flip(mk, 0)
            k = random.randint(0, 3)
            if k:
                im, mk = np.rot90(im, k).copy(), np.rot90(mk, k).copy()
            if random.random() < 0.3:
                a, b = random.uniform(0.9, 1.1), random.uniform(-12, 12)
                im = np.clip(im.astype(np.float32) * a + b, 0, 255).astype(np.uint8)

        x = torch.from_numpy(np.ascontiguousarray(im.transpose(2, 0, 1)).astype(np.float32) / 255.0)
        y = torch.from_numpy(np.ascontiguousarray(mk)[None].astype(np.float32))
        return x, y, r["image"].name


# ------------------------------------------------------------------ 5. 网络
class DoubleConv(nn.Module):
    def __init__(self, cin, cout):
        super().__init__()
        self.block = nn.Sequential(
            nn.Conv2d(cin, cout, 3, padding=1, bias=False), nn.BatchNorm2d(cout), nn.ReLU(inplace=True),
            nn.Conv2d(cout, cout, 3, padding=1, bias=False), nn.BatchNorm2d(cout), nn.ReLU(inplace=True))

    def forward(self, x):
        return self.block(x)


class UNet(nn.Module):
#     """4 级下采样：512 → 256 → 128 → 64 → 32，再逐级上采样回 512。输出 logits。"""
    def __init__(self, cin=3, base=32):
        super().__init__()
        self.d1 = DoubleConv(cin, base)
        self.d2 = DoubleConv(base, base * 2)
        self.d3 = DoubleConv(base * 2, base * 4)
        self.d4 = DoubleConv(base * 4, base * 8)
        self.pool = nn.MaxPool2d(2)
        self.u3 = nn.ConvTranspose2d(base * 8, base * 4, 2, stride=2)
        self.c3 = DoubleConv(base * 8, base * 4)
        self.u2 = nn.ConvTranspose2d(base * 4, base * 2, 2, stride=2)
        self.c2 = DoubleConv(base * 4, base * 2)
        self.u1 = nn.ConvTranspose2d(base * 2, base, 2, stride=2)
        self.c1 = DoubleConv(base * 2, base)
        self.head = nn.Conv2d(base, 1, 1)

    def forward(self, x):
        x1 = self.d1(x)                                  # 512, 32 通道
        x2 = self.d2(self.pool(x1))                      # 256, 64
        x3 = self.d3(self.pool(x2))                      # 128, 128
        x4 = self.d4(self.pool(x3))                      # 64, 256
        y3 = self.c3(torch.cat([self.u3(x4), x3], 1))    # 128
        y2 = self.c2(torch.cat([self.u2(y3), x2], 1))    # 256
        y1 = self.c1(torch.cat([self.u1(y2), x1], 1))    # 512
        return self.head(y1)


# ---------------------- 6. 损失与指标
class BCEDiceLoss(nn.Module):
#     """裂缝只占 10.8%，单用 BCE 网络会偷懒全判背景，Dice 专门治这个。"""
    def __init__(self):
        super().__init__()
        self.bce = nn.BCEWithLogitsLoss()

    def forward(self, logits, target):
        bce = self.bce(logits, target)
        prob = torch.sigmoid(logits).float()             # 半精度下先把 dice 拉到 fp32
        tgt = target.float()
        dims = (1, 2, 3)
        inter = (prob * tgt).sum(dims)
        dice = (2.0 * inter + 1.0) / (prob.sum(dims) + tgt.sum(dims) + 1.0)
        return bce + (1.0 - dice.mean())


def metrics_from_counts(tp, fp, fn, tn):
    eps = 1e-8
    return {"IoU": tp / (tp + fp + fn + eps),
            "precision": tp / (tp + fp + eps),
            "recall": tp / (tp + fn + eps),
            "TP": tp, "FP": fp, "FN": fn, "TN": tn}


def run_epoch(model, loader, criterion, dev, optimizer=None, scaler=None):
    training = optimizer is not None
    model.train(training)
    amp = dev.type == "cuda"
    total, tp = 0.0, 0; fp = fn = tn = 0

    for x, y, _ in loader:
        x, y = x.to(dev, non_blocking=True), y.to(dev, non_blocking=True)
        if training:
            optimizer.zero_grad(set_to_none=True)
        with torch.set_grad_enabled(training):
            # CPU 上不开半精度：torch 的 CPU autocast 只认 bfloat16
            ctx = torch.autocast(device_type="cuda", dtype=torch.float16) if amp else contextlib.nullcontext()
            with ctx:
                logits = model(x)
                loss = criterion(logits, y)
            if training:
                if amp and scaler is not None:
                    scaler.scale(loss).backward(); scaler.step(optimizer); scaler.update()
                else:
                    loss.backward(); optimizer.step()

        with torch.no_grad():
            pred = (torch.sigmoid(logits.float()) >= 0.5)
            gt = (y >= 0.5)
            tp += int((pred & gt).sum()); fp += int((pred & ~gt).sum())
            fn += int((~pred & gt).sum()); tn += int((~pred & ~gt).sum())
        total += float(loss.item()) * x.size(0)

    out = metrics_from_counts(tp, fp, fn, tn)
    out["loss"] = total / max(len(loader.dataset), 1)
    return out


@torch.no_grad()
def save_panels(model, records, dev, out_dir: Path, limit=SAVE_PANELS):
#     """存三联图：左原图｜中真值｜右预测。课上直接投这一组。"""
    pdir = out_dir / "test_predictions"
    pdir.mkdir(parents=True, exist_ok=True)
    model.eval()
    for r in list(records)[:limit]:
        im = cv2.imread(str(r["image"]), cv2.IMREAD_COLOR)
        mk = cv2.imread(str(r["mask"]), cv2.IMREAD_GRAYSCALE)
        rgb = cv2.cvtColor(im, cv2.COLOR_BGR2RGB)
        t = torch.from_numpy(rgb.transpose(2, 0, 1).astype(np.float32) / 255.0)[None].to(dev)
        prob = torch.sigmoid(model(t))[0, 0].cpu().numpy()
        pred = (prob >= 0.5).astype(np.uint8) * 255
        gt = ((mk > 0).astype(np.uint8)) * 255

        overlay = rgb.copy()
        m = pred > 0
        overlay[m] = (0.55 * overlay[m] + 0.45 * np.array([255, 40, 40])).astype(np.uint8)

        panel = np.concatenate([
            cv2.cvtColor(rgb, cv2.COLOR_RGB2BGR),
            cv2.cvtColor(gt, cv2.COLOR_GRAY2BGR),
            cv2.cvtColor(pred, cv2.COLOR_GRAY2BGR),
            cv2.cvtColor(overlay, cv2.COLOR_RGB2BGR)], axis=1)
        cv2.imwrite(str(pdir / ("%s_panel.png" % r["image"].stem)), panel)


# --------------------- 7. 主流程
def main():
    args = parse_args()
    assert args.size % 16 == 0
    set_seed(args.seed)
    root, out_dir = Path(args.data), Path(args.out)
    out_dir.mkdir(parents=True, exist_ok=True)

    dev = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    print("[设备] %s" % (torch.cuda.get_device_name(0) if dev.type == "cuda" else "CPU（会慢，建议 --size 256"))

    pairs = collect_pairs(root)
    tr, va, te = stratified_split(pairs, args.seed)
    print("[划分] 训练 %d / 验证 %d / 测试 %d（按裂缝占比分层）" % (len(tr), len(va), len(te)))
    save_split(tr, va, te, out_dir)

    dl_tr = DataLoader(CrackDataset(tr, args.size, not args.no_aug), batch_size=args.batch,
                       shuffle=True, num_workers=args.workers, pin_memory=(dev.type == "cuda"))
    dl_va = DataLoader(CrackDataset(va, args.size), batch_size=args.batch, num_workers=args.workers)
    dl_te = DataLoader(CrackDataset(te, args.size), batch_size=args.batch, num_workers=args.workers)

    model = UNet(3, 32).to(dev)
    opt = torch.optim.AdamW(model.parameters(), lr=args.lr, weight_decay=1e-4)
    sch = torch.optim.lr_scheduler.CosineAnnealingLR(opt, T_max=args.epochs, eta_min=args.lr * 0.05)
    crit = BCEDiceLoss()
    scaler = torch.cuda.amp.GradScaler(enabled=(dev.type == "cuda"))

    best, log = -1.0, []
    for ep in range(1, args.epochs + 1):
        t0 = time.time()
        m_tr = run_epoch(model, dl_tr, crit, dev, opt, scaler)
        m_va = run_epoch(model, dl_va, crit, dev)
        sch.step()
        print("epoch %02d/%02d  train loss %.4f IoU %.3f | val loss %.4f IoU %.3f P %.3f R %.3f | %ds"
              % (ep, args.epochs, m_tr["loss"], m_tr["IoU"], m_va["loss"], m_va["IoU"],
                 m_va["precision"], m_va["recall"], int(time.time() - t0)))
        log.append([ep, m_tr["loss"], m_tr["IoU"], m_va["loss"], m_va["IoU"],
                    m_va["precision"], m_va["recall"], int(time.time() - t0)])

        torch.save(model.state_dict(), out_dir / "last_unet.pt")
        if m_va["IoU"] > best:                    # 只看验证集，不看训练 loss
            best = m_va["IoU"]
            torch.save({"model": model.state_dict(), "epoch": ep, "val_iou": best},
                       out_dir / "best_unet.pt")
            print("        ↑ 新的最好，已存 best_unet.pt")

    with (out_dir / "train_log.csv").open("w", encoding="utf-8-sig", newline="") as f:
        w = csv.writer(f)
        w.writerow(["epoch", "train_loss", "train_iou", "val_loss", "val_iou", "val_precision", "val_recall", "seconds"])
        w.writerows(log)

    # 期末测评：用验证集选出的那一版，测试集只考这一次
    ck = torch.load(out_dir / "best_unet.pt", map_location=dev)
    model.load_state_dict(ck["model"])
    m_te = run_epoch(model, dl_te, crit, dev)
    print("\n[测试] 载入第 %d 轮（val_IoU %.4f），在 %d 张上只考一次"
          % (ck["epoch"], ck["val_iou"], len(te)))
    print("TP %d  FP %d  FN %d  TN %d" % (m_te["TP"], m_te["FP"], m_te["FN"], m_te["TN"]))
    print("IoU %.4f  Precision %.3f  Recall %.3f" % (m_te["IoU"], m_te["precision"], m_te["recall"]))

    result = {"checkpoint_epoch": int(ck["epoch"]),
              "counts": {"train": len(tr), "val": len(va), "test": len(te)},
              "test_metrics": m_te, "threshold": 0.5,
              "warning": "结果仅对这一次划分有效；不能直接当作毫米精度或别的桥的性能。"}
    with (out_dir / "test_metrics.json").open("w", encoding="utf-8") as f:
        json.dump(result, f, ensure_ascii=False, indent=2)

    save_panels(model, te, dev, out_dir)
    print("三联图已存到 %s" % (out_dir / "test_predictions"))


if __name__ == "__main__":
    main()
